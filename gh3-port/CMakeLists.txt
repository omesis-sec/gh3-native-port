cmake_minimum_required(VERSION 3.21)
project(GuitarHero3NativePrototype LANGUAGES C CXX)

# This prototype expects locally generated code and the local PS2Recomp checkout.
set(PS2X_BUILD_RECOMP OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_ANALYZER OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_RUNTIME ON CACHE BOOL "" FORCE)
set(PS2X_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_STUDIO OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_DEBUG_UI OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_FFMPEG OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_AGRESSIVE_LOGS OFF CACHE BOOL "" FORCE)
set(PS2X_RUNNER_UNITY_BUILD_BATCH_SIZE 16 CACHE STRING "" FORCE)
set(GH3_GENERATED_DIR "${CMAKE_SOURCE_DIR}/local/hybrid" CACHE PATH "Locally generated game sources")
if(NOT EXISTS "${GH3_GENERATED_DIR}/register_functions.cpp")
    message(FATAL_ERROR "Generate the hybrid sources in gh3-port/local/hybrid first")
endif()
set(PS2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_subdirectory("${PS2_ROOT}" "${CMAKE_BINARY_DIR}/upstream")
# Replace the upstream empty table with the table emitted for this game.
get_target_property(GH3_RUNNER_SOURCES ps2EntryRunner SOURCES)
list(FILTER GH3_RUNNER_SOURCES EXCLUDE REGEX "[/\\]src[/\\]runner[/\\]register_functions[.]cpp$")
set_property(TARGET ps2EntryRunner PROPERTY SOURCES "${GH3_RUNNER_SOURCES}")
file(GLOB GH3_GENERATED_SOURCES CONFIGURE_DEPENDS "${GH3_GENERATED_DIR}/*.cpp")
set_source_files_properties(src/diagnostic_logging.cpp PROPERTIES SKIP_UNITY_BUILD_INCLUSION ON)
target_sources(ps2EntryRunner PRIVATE ${GH3_GENERATED_SOURCES} src/diagnostic_logging.cpp)
target_include_directories(ps2EntryRunner BEFORE PRIVATE "${GH3_GENERATED_DIR}")
add_executable(gh3_iop_memory_test tests/iop_memory_reuse.cpp)
target_link_libraries(gh3_iop_memory_test PRIVATE ps2_iop)
target_include_directories(gh3_iop_memory_test PRIVATE "${PS2_ROOT}/ps2xIOP/src")
if(MSVC)
    # MSBuild's project-level parallelism alone does not parallelize source files.
    target_compile_options(ps2_runtime PRIVATE /MP4)
    target_compile_options(ps2EntryRunner PRIVATE /MP4)
endif()
